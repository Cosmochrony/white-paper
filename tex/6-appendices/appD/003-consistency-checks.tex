\subsection{Order-of-Magnitude Consistency Checks}
\label{sec:consistency_checks}

All quantities below use the convention of Appendix~\ref{subsec:fundamental-quantities}
($\chi_c$ a length, $K_0$ of dimension $\mathrm{s}^{-2}$).
\begin{enumerate}
  \item \textbf{Electron mass scale.}
    The lowest stability eigenvalue must satisfy
    $\lambda_1 \sim (m_e c^2/\hbar_{\mathrm{eff}})^2
      \approx 6\times 10^{41}\,\mathrm{s^{-2}}$.
    No relation between $\lambda_1$, $K_0$ and the lattice spacing $a$ is established in this paper, so this target
    does not fix $K_0$.
  \item \textbf{Correlation scale.}
    The proportionality $\langle\phi_H\rangle \propto \hbar_{\mathrm{eff}}c/\chi_c$ of
    Section~\ref{subsec:chi_c-electroweak-scale}, with a coefficient of order one, calibrates
    $\chi_c \sim \hbar c/v \approx 8\times 10^{-19}\,\mathrm{m}$.
    This is a calibration, not a bound.
\end{enumerate}
